Hunt Query of the Week

Detecting unauthorized agent activity often requires monitoring for unusual command execution patterns or unexpected network connections. This query targets suspicious PowerShell activity that might indicate an agent attempting to exfiltrate data or pivot. Adapt the `ProcessName` and `CommandLine` arguments based on your environment and known agent behaviors.

let suspicious_commands = dynamic(['Invoke-WebRequest', 'Invoke-RestMethod', 'Get-ChildItem', 'Get-Content', 'Copy-Item', 'Move-Item', 'Start-Process', 'iex', 'Invoke-Expression']);
DeviceProcessEvents
| where Timestamp > ago(7d)
| where FileName =~ "powershell.exe"
| mv-flatten CommandLine
| where CommandLine contains any (suspicious_commands)
| extend CommandArgs = split(CommandLine, ' ')
| where array_length(CommandArgs) > 1
| project Timestamp, DeviceName, InitiatingProcessFileName, FileName, CommandLine, CommandArgs
| summarize count() by DeviceName, InitiatingProcessFileName, FileName, bin(Timestamp, 1h)
| where count_ > 5 // Adjust threshold for noise
| project Timestamp, DeviceName, InitiatingProcessFileName, FileName, CommandLine
| order by Timestamp desc
